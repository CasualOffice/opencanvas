  • GET /api/v1/openapi.json

This document

GET/api/v1/openapi.json

The OpenAPI 3.1 document of this server's API, the same file as crates/oc-server/api/openapi.json in the repository. Public: no credentials, not rate limited.

Credentials
None: the document is public.
Operation id
getOpenApi

Responses

200OK

Body application/json

object

Example

{
  "openapi": "3.1.0",

    "title": "OpenCanvas headless API",

Request
curl -s "https://canvas.example.com/api/v1/openapi.json"
import { OpenCanvas } from "@opencanvas/client";

const oc = new OpenCanvas({
  baseUrl: "https://canvas.example.com",
  auth: { hostId: "acme", key: process.env.OC_API_KEY! },
});

const result = await oc.getOpenApi();
console.log(result);
use opencanvas_client::{Auth, CallOptions, Client, Params};

let oc = Client::builder("https://canvas.example.com")
    .auth(Auth::host_key("acme", std::env::var("OC_API_KEY").expect("OC_API_KEY")))
    .build();

let res = oc
    .call(
        "getOpenApi",
        &Params::new(),
        &CallOptions::default(),
    )
    .await?;
let answer: serde_json::Value = res.value()?;
println!("{answer:#}");
import requests

r = requests.get(
    "https://canvas.example.com/api/v1/openapi.json",
    timeout=30,
)
r.raise_for_status()
print(r.json())
